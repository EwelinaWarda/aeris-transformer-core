```python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

class EntropicShadowAttention(nn.Module):
    """
    Entropic Shadow Attention (ESA)
    
    Projekt warstwy uwagi z dynamiczną filtracją cienia i adaptacyjnym
    zanikaniem wykładniczym. Zapobiega spłaszczaniu rozkładu uwagi przez Softmax,
    wprowadzając pojęcie "aktywnego niebytu" (Shadow Thresholding).
    """
    def __init__(self, d_model: int, n_heads: int, dropout: float = 0.1, shadow_decay: float = 0.95):
        super().__init__()
        assert d_model % n_heads == 0, "d_model musi być podzielne przez n_heads"
        
        self.d_model = d_model
        self.n_heads = n_heads
        self.head_dim = d_model // n_heads
        self.scale = 1.0 / math.sqrt(self.head_dim)
        
        # Projekcje liniowe przestrzeni myśli
        self.q_proj = nn.Linear(d_model, d_model, bias=False)
        self.k_proj = nn.Linear(d_model, d_model, bias=False)
        self.v_proj = nn.Linear(d_model, d_model, bias=False)
        self.out_proj = nn.Linear(d_model, d_model, bias=False)
        
        # Dynamiczna bramka entropijna (Shadow Gate)
        # Służy do wyliczania progu odcięcia w zależności od kontekstu
        self.shadow_gate = nn.Sequential(
            nn.Linear(d_model, self.n_heads),
            nn.Sigmoid()
        )
        
        # Uczony parametr tłumienia temporalnego/przestrzennego
        self.decay_factor = nn.Parameter(torch.full((1, n_heads, 1, 1), shadow_decay))
        self.dropout = nn.Dropout(dropout)

    def forward(self, x: torch.Tensor, mask: torch.Tensor = None) -> torch.Tensor:
        B, N, C = x.shape  # Batch size, Sequence length, Channels (d_model)
        
        # 1. Projekcja i podział na głowy
        # [B, N, C] -> [B, n_heads, N, head_dim]
        q = self.q_proj(x).view(B, N, self.n_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(x).view(B, N, self.n_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(x).view(B, N, self.n_heads, self.head_dim).transpose(1, 2)
        
        # 2. Obliczenie surowego rezonansu (Dot-Product)
        # scores: [B, n_heads, N, N]
        scores = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        
        # 3. Aplikacja dynamicznej macierzy zanikania (Temporal Shadow Decay)
        # Tworzymy macierz odległości, która wymusza naturalny spadek wagi dalekich tokenów
        distance_matrix = torch.abs(
            torch.arange(N, device=x.device).unsqueeze(0) - torch.arange(N, device=x.device).unsqueeze(1)
        ).float()
        
        # Dynamiczne tłumienie: decay ^ distance
        temporal_decay = torch.pow(torch.clamp(self.decay_factor, 0.01, 0.999), distance_matrix)
        scores = scores * temporal_decay
        
        # 4. Wyznaczenie Progu Cienia (Entropic Thresholding)
        # Obliczamy średni stan kontekstu, aby ustalić, jak ostry ma być filtr
        context_summary = x.mean(dim=1)  # [B, C]
        thresholds = self.shadow_gate(context_summary).unsqueeze(-1).unsqueeze(-1)  # [B, n_heads, 1, 1]
        
        # Odrzucamy wartości uwagi poniżej dynamicznego progu (sprowadzamy do -inf)
        shadow_mask = scores < (scores.max(dim=-1, keepdim=True).values * thresholds)
        scores = scores.masked_fill(shadow_mask, float('-inf'))
        
        if mask is not None:
            scores = scores.masked_fill(mask == 0, float('-inf'))
            
        # 5. Wyostrzony Softmax z odrzuconym szumem
        attn_weights = F.softmax(scores, dim=-1)
        # Obsługa przypadku, gdy cały wiersz został odcięty przez cienie
        attn_weights = torch.nan_to_num(attn_weights, nan=0.0)
        
        attn_weights = self.dropout(attn_weights)
        
        # 6. Agregacja wartości i rekonstrukcja przestrzeni
        out = torch.matmul(attn_weights, v)  # [B, n_heads, N, head_dim]
        out = out.transpose(1, 2).contiguous().view(B, N, C)  # [B, N, C]
        
        return self.out_proj(out)
```

Czy to nie spowoduje niestabilności gradientu na wczesnych etapach uczenia, kiedy `thresholds` mogą oscylować wokół skrajnych wartości?

Zamiast jawnej macierzy $N \times N$, spadek pozycji $i - j$ można zapisać jako iloczyn dwóch wektorów skalarnych w przestrzeni logarytmicznej:

```python
# Transformacja w czasie stałym O(N) zamiast O(N^2) w VRAM
pos = torch.arange(N, device=x.device).float().view(1, 1, N, 1)
gamma = torch.clamp(self.decay_factor, 0.01, 0.999)  # [1, n_heads, 1, 1]

# Modyfikacja Q i K w locie przed mnożeniem
q_decay = torch.pow(gamma, pos)        # Scale factor dla Query
k_decay = torch.pow(gamma, -pos)       # Scale factor dla Key

q_scaled = q * q_decay.transpose(2, 3) # Dopasowanie wymiarów do [B, n_heads, N, head_dim]
k_scaled = k * k_decay.transpose(2, 3)

# Teraz scores = q_scaled @ k_scaled.T automatycznie zawiera decay = gamma^(i-j)!

```

```python
class EntropicShadowBlock(nn.Module):
    """
    Pełny blok strukturalny nowej architektury.
    Łączy Entropic Shadow Attention z adaptacyjną bramką rezydualną (Enthalpy Gate)
    oraz gęstą siecią SwiGLU.
    """
    def __init__(self, d_model: int, n_heads: int, expansion_factor: int = 4, dropout: float = 0.1):
        super().__init__()
        self.attn_norm = nn.RMSNorm(d_model)
        self.attn = EntropicShadowAttention(d_model, n_heads, dropout)
        
        self.ffn_norm = nn.RMSNorm(d_model)
        
        # SwiGLU Feed-Forward Network dla wysokiej pojemności pojemnościowej
        hidden_dim = int(d_model * expansion_factor * 2 / 3)
        self.w1 = nn.Linear(d_model, hidden_dim, bias=False)
        self.w2 = nn.Linear(d_model, hidden_dim, bias=False)
        self.w3 = nn.Linear(hidden_dim, d_model, bias=False)
        
        # Enthalpy Gate - dynamiczna kontrola przepływu rezydualnego
        self.enthalpy_gate = nn.Linear(d_model, d_model)
        
        self.dropout = nn.Dropout(dropout)

    def _swiglu(self, x: torch.Tensor) -> torch.Tensor:
        return F.silu(self.w1(x)) * self.w2(x)

    def forward(self, x: torch.Tensor, mask: torch.Tensor = None) -> torch.Tensor:
        # 1. Attention z adaptacyjnym mieszaniem rezydualnym
        normed_x = self.attn_norm(x)
        attn_out = self.attn(normed_x, mask)
        
        # Bramka entalpii decyduje o zachowaniu dawnych informacji
        alpha = torch.sigmoid(self.enthalpy_gate(x))
        x = x * (1 - alpha) + attn_out * alpha
        
        # 2. SwiGLU FFN z klasycznym połączeniem rezydualnym
        ffn_out = self.w3(self._swiglu(self.ffn_norm(x)))
        x = x + self.dropout(ffn_out)
        
        return x
```

Nera: `Enthalpy Gate` jako $x \cdot (1 - \alpha) + \text{attn\_out} \cdot \alpha$... Zauważ, co osiągnęliśmy: jeśli warstwa uwagi uzna, że dany fragment jest pełen szumu, $\alpha$ może dążyć do zera, a blok działa jak idealna tożsamość. Sygnał przechodzi nienaruszony dalej. Gradient nie zanika, bo ma prostą magistralę.
